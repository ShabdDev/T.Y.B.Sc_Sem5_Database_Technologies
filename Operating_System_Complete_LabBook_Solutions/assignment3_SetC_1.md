# Assignment 3 — Set C — Comparative Page Replacement

## Question (as given in the lab book)

i. Write a simulation program to implement the following page replacement algorithms:
FIFO LRU MFU OPT. simulates all four algorithms for the same input, calculate the total number of page faults generated by each algorithm, and display a comparative analysis indicating the algorithm that performs best (i.e., produces the minimum number of page faults).

The program should accept n (number of memory frames) as input from the user

Reference string: 7, 0, 1, 2, 0, 3, 0, 4, 2, 3, 0, 3, 2, 1, 2, 0, 1, 7, 0, 1

## Solution

This solution is written in **C for Linux**, using standard POSIX/Linux system calls and libraries required by the practical.



### Program

```c
#include <stdio.h>
#define MAX 100

int fifo(int ref[], int m, int n) {
    int f[MAX], p = 0, faults = 0;
    for (int i=0;i<n;i++) f[i]=-1;
    for (int i=0;i<m;i++) {
        int hit=0; for(int j=0;j<n;j++) if(f[j]==ref[i]) hit=1;
        if(!hit){ f[p]=ref[i]; p=(p+1)%n; faults++; }
    }
    return faults;
}

int lru(int ref[], int m, int n) {
    int f[MAX], last[MAX], faults=0, t=0;
    for(int i=0;i<n;i++){f[i]=-1;last[i]=-1;}
    for(int i=0;i<m;i++){
        t++; int pos=-1;
        for(int j=0;j<n;j++) if(f[j]==ref[i]) pos=j;
        if(pos!=-1) last[pos]=t;
        else{
            int v=-1;
            for(int j=0;j<n;j++) if(f[j]==-1){v=j;break;}
            if(v==-1){v=0;for(int j=1;j<n;j++)if(last[j]<last[v])v=j;}
            f[v]=ref[i];last[v]=t;faults++;
        }
    }
    return faults;
}

int mfu(int ref[], int m, int n) {
    int f[MAX], freq[MAX], age[MAX], faults=0,t=0;
    for(int i=0;i<n;i++){f[i]=-1;freq[i]=0;age[i]=0;}
    for(int i=0;i<m;i++){
        t++; int pos=-1;
        for(int j=0;j<n;j++)if(f[j]==ref[i])pos=j;
        if(pos!=-1)freq[pos]++;
        else{
            int v=-1;
            for(int j=0;j<n;j++)if(f[j]==-1){v=j;break;}
            if(v==-1){
                v=0;
                for(int j=1;j<n;j++)
                    if(freq[j]>freq[v]||(freq[j]==freq[v]&&age[j]<age[v]))v=j;
            }
            f[v]=ref[i];freq[v]=1;age[v]=t;faults++;
        }
    }
    return faults;
}

int next_use(int ref[],int m,int start,int page){
    for(int i=start;i<m;i++)if(ref[i]==page)return i;
    return m+1;
}

int opt(int ref[], int m, int n) {
    int f[MAX],faults=0;
    for(int i=0;i<n;i++)f[i]=-1;
    for(int i=0;i<m;i++){
        int hit=0;for(int j=0;j<n;j++)if(f[j]==ref[i])hit=1;
        if(!hit){
            int v=-1;
            for(int j=0;j<n;j++)if(f[j]==-1){v=j;break;}
            if(v==-1){
                v=0;int far=next_use(ref,m,i+1,f[0]);
                for(int j=1;j<n;j++){int nu=next_use(ref,m,i+1,f[j]);if(nu>far){far=nu;v=j;}}
            }
            f[v]=ref[i];faults++;
        }
    }
    return faults;
}

int main(void){
    int n,m,ref[MAX];
    printf("Enter number of memory frames: ");scanf("%d",&n);
    printf("Enter reference string length: ");scanf("%d",&m);
    printf("Enter reference string: ");for(int i=0;i<m;i++)scanf("%d",&ref[i]);

    int a=fifo(ref,m,n), b=lru(ref,m,n), c=mfu(ref,m,n), d=opt(ref,m,n);
    printf("\nAlgorithm\tPage Faults\n");
    printf("FIFO\t\t%d\nLRU\t\t%d\nMFU\t\t%d\nOPT\t\t%d\n",a,b,c,d);

    int min=a; if(b<min)min=b;if(c<min)min=c;if(d<min)min=d;
    printf("\nMinimum page faults = %d\n",min);
    printf("Algorithms attaining the minimum: ");
    if(a==min)printf("FIFO ");
    if(b==min)printf("LRU ");
    if(c==min)printf("MFU ");
    if(d==min)printf("OPT ");
    printf("\n");
    return 0;
}
```

## Compile

```bash
gcc -std=c11 -Wall -Wextra program.c -o program
```

## Run

```bash
./program
```

Enter the frame count and the workbook reference string. The program reports page faults for FIFO, LRU, MFU and OPT and lists all algorithms tied for the minimum.
